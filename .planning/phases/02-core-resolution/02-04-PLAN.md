```typescript
import type {
  ThemeManifest,
  ResolvedTheme,
  ThemeMode,
} from "@mosaix/contracts";
import { ThemeNotFoundError, ThemeCycleError } from "./theme-errors";
import type { ThemeInheritanceResolverLike } from "./theme-resolver";

/** Manifest lookup seam — injected, mirrors MigrationPlanner constructor style. */
export interface ThemeManifestLookup {
  get(themeId: string): ThemeManifest | undefined;
}

export interface ThemeInheritanceResolverOptions {
  readonly lookup: ThemeManifestLookup;
}

export class ThemeInheritanceResolver implements ThemeInheritanceResolverLike {
  constructor(options: ThemeInheritanceResolverOptions);

  /** DFS parent-first resolve of the extends graph rooted at themeId. */
  resolve(
    themeId: string,
    mode: ThemeMode,
    load: (id: string) => ThemeManifest | undefined,
  ): ResolvedTheme;
}

/** Standalone helper for tests/consumers — resolves an extends chain to a merged ResolvedTheme. */
export function resolveThemeInheritance(
  themeId: string,
  mode: ThemeMode,
  load: (id: string) => ThemeManifest | undefined,
): ResolvedTheme;
```

Merge rule (exact): recursion returns the parent's merged `ResolvedTheme`;
the child overlays its `manifest` last. Concrete shape for a leaf `id` with
child manifest `childManifest` and parent resolution `parentMerged`:
`resolved = { themeId: id, version: childManifest.version, mode,
manifest: deepMerge(parentMerged.manifest, childManifest) }` — the merged
`manifest` carries parent base + child overlay with a **recursive** merge:
child scalars and arrays win, nested plain-object subtrees (`tokens`,
`modes`, `accessibility`, `assets`, `metadata`) merge depth-first so a
partial child overlay preserves the parent's remaining subtree (CR-01),
`themeId` is the LEAF id, leaf `version`/`extends` preserved at the
ResolvedTheme level. Cycle path: the DFS stack at the moment the repeated node
is re-entered, e.g. `["a", "b", "a"]` → `ThemeCycleError(["a","b","a"])`.
