```typescript
// PRD §4.2b — capabilities per target type.
export interface ThemeTargetCapabilities {
  readonly userSelectable: boolean;
  readonly adminConfigurable: boolean;
}

// PRD §4.2b — a registration is type + capabilities.
export interface ThemeTargetRegistration {
  readonly type: string;
  readonly capabilities: ThemeTargetCapabilities;
}

// The registry: single source of target-type declarations.
// Registration is bootstrap-only (D-16): the resolver never calls register().
export class ThemeTargetRegistry {
  private readonly types = new Map<string, ThemeTargetRegistration>();

  constructor(initial?: readonly ThemeTargetRegistration[]);

  /** Programmatic surface + the mutation shared by both surfaces. last-write-wins (no throw — capability-registry convention). */
  register(registration: ThemeTargetRegistration): void;

  get(type: string): ThemeTargetRegistration | undefined; // undefined for unknown (CONVENTIONS: lookups return undefined)
  has(type: string): boolean; // this.get(type) !== undefined
  list(): ThemeTargetRegistration[]; // Array.from(map.values())
  get size(): number; // export of map size
}
```

## Trust Boundaries

| Boundary                        | Description                                                                                                 |
| ------------------------------- | ----------------------------------------------------------------------------------------------------------- |
| project config / SDK → registry | Untrusted type declarations enter the kernel at bootstrap via declarative list or programmatic `register()` |

## STRIDE Threat Register

| Threat ID       | Category               | Component                               | Disposition | Mitigation Plan                                                                                                                                                          |
| --------------- | ---------------------- | --------------------------------------- | ----------- | ------------------------------------------------------------------------------------------------------------------------------------------------------------------------ |
| T2-01           | Tampering              | `ThemeTargetRegistry.register()`        | mitigate    | Last-write-wins WITHOUT throw (capability-registry convention) — a duplicate/ hostile re-register silently occupies, never corrupts state; deterministic single Map      |
| T2-02           | Tampering              | `ThemeTargetRegistration./capabilities` | mitigate    | Capability flags are plain booleans with no default — a missing flag yields `false`; registry stores the exact object passed; tests assert exact shape via expectType-Of |
| T2-03           | Information disclosure | `register()` payload                    | accept      | registry stores types/capabilities ONLY (strings + booleans); attacker-supplied extra fields are ignored because assignments are typed — no logger, no data leak surface |
| T2-04           | Denial of service      | unbounded `register()`                  | accept      | Bootstrap-only registration (D-16) bounds the count; the kernel Phase 3 `ThemeRuntime` bootstrap phase controls who calls register; no rate-limit surface in Phase 2     |
| T2-05           | Elevation of privilege | N/A                                     | accept      | No authorization surface in this registry (auth policy is Governance/PolicyResolver, PRD §7); registry is a pure declaration store                                       |
| </threat_model> |
