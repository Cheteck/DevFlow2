import { KernelError } from "../kernel-errors";

// D-22: theme-prefixed code union (no string-matching on messages)
export type ThemeErrorCode =
  | "THEME_NOT_FOUND"
  | "THEME_CYCLE"
  | "THEME_VERSION"
  | "THEME_VALIDATION"
  | "THEME_COMPATIBILITY"
  | "THEME_AUTHORIZATION";

export class ThemeError extends KernelError {
  constructor(
    code: ThemeErrorCode,
    message: string,
    details: KernelErrorDetails = {},
  ) {
    super(code, message, details);
    this.name = "ThemeError";
  }
}

// Phase-2 RAISED errors (D-21). Typed-args + structured details (see registration error style);
// THEME_COMPATIBILITY / THEME_AUTHORIZATION are declared but NOT raised here ('// (Phase 3) gated' comment)
export class ThemeNotFoundError extends ThemeError {
  constructor(themeId: string) {
    super("THEME_NOT_FOUND", `Theme not found: ${themeId}`, { themeId });
    this.name = "ThemeNotFoundError";
  }
}

export class ThemeCycleError extends ThemeError {
  constructor(cycle: readonly string[]) {
    super(
      "THEME_CYCLE",
      `Theme inheritance cycle detected: ${cycle.join(" → ")}`,
      { cycle },
    );
    this.name = "ThemeCycleError";
  }
}

export class ThemeVersionError extends ThemeError {
  constructor(themeId: string, required: string, found: string) {
    super(
      "THEME_VERSION",
      `Theme ${themeId}: required ${required} but found ${found}`,
      { themeId, required, found },
    );
    this.name = "ThemeVersionError";
  }
}

export class ThemeValidationError extends ThemeError {
  constructor(issues: readonly unknown[]) {
    super("THEME_VALIDATION", "Invalid theme assignment", { issues });
    this.name = "ThemeValidationError";
  }
}

import type {
  ThemeAssignment,
  ThemeTarget,
} from "./theme-assignment-type-reexport"; // ← see note below

export interface ThemeAssignmentsStore {
  /** Persists an assignment (create or replace by target key). */
  assign(assignment: ThemeAssignment): void;
  /** Removes the assignment for the target; idempotent no-op if absent. */
  unassign(target: ThemeTarget): void;
  /** Returns the assignment for the target, or undefined when absent. */
  get(target: ThemeTarget): ThemeAssignment | undefined;
  /** All assignments as a snapshot copy. */
  list(): ThemeAssignment[];
}

import type {
  ThemeAssignment,
  ThemeTarget,
  ThemeAssignmentsStore,
} from "@mosaix/contracts";
import type { ThemeAssignmentChangedPayload } from "@mosaix/contracts";

export interface ThemeMutationListener {
  (mutation: ThemeAssignmentChangedPayload): void;
}

export interface InMemoryThemeAssignmentsStoreOptions {
  /** Called synchronously after assign()/unassign() with the mutation payload. */
  onMutation?: ThemeMutationListener;
}

export class InMemoryThemeAssignmentsStore implements ThemeAssignmentsStore {
  constructor(options: InMemoryThemeAssignmentsStoreOptions = {});
  // key = `${target.type}:${target.id}` — same composite style as EventStore/EventBus
  assign(assignment: ThemeAssignment): void; // validates via ThemeAssignmentSchema (strict — reuse @mosaix/schemas), stores, then emits { target, assignment, changedBy: assignment.updatedBy ?? "system", at: assignment.updatedAt ?? new Date().toISOString() }
  unassign(target: ThemeTarget): void; // removal when present -> emit with the removed assignment record; absent -> no-op
  get(target: ThemeTarget): ThemeAssignment | undefined; // returns stored object by key (or undefined)
  list(): ThemeAssignment[]; // Array.from(map.values()) snapshot copy
}

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|------------|------------------|
| T-02-06 | Tampering | `ThemeAssignmentsStore.assign()` | mitigate | Strict Zod validation of the whole assignment (`ThemeAssignmentSchema.safeParse`) BEFORE the write; malformed → `ThemeValidationError` with issues (structural, no mutation) |
| T-02-07 | Information disclosure | `ThemeValidationError.issues` | accept | `issues` surfaces path/message of the failing field only; quoted `data` errors are zod-standard typed — no secrets by construction (assignment carries only targetId/themeId/mode/version/source) |
| T-02-08 | Spoofing | `ThemeAssignment.source` in emission payload | mitigate | `changedBy` default is `updatedBy ?? "system"`; the source/audit trail is preserved in the emitted payload; await further Policy/GoD service for true auth in Phase 5 |
| T-02-09 | Denial of service | unbounded `list()`/`Map` growth | accept | In-memory by design (D-18 persists later); bounded by assignment count of a single target; Phase-2 only |
| T-02-10 | Repudiation | removed records on `unassign()` | accept | Emitted payload carries the removed assignment + `at` timestamp which the service layer (02-04) routes to event bus / audit; store retains no permanent log in-memory (out of scope) |
</threat_model>
