```typescript
// ————— NEW FILE: packages/core/src/theme/theme-cache.ts —————
import type { CompiledTheme } from "@mosaix/contracts";

/** Default LRU cap (~100 entries, 03-CONTEXT). */
export const THEME_CACHE_DEFAULT_MAX = 100;

/** Opaque cache key type — composed `themeId:version:mode`. */
export type ThemeCacheKey = string;

/** Compose the canonical key. */
export function toThemeCacheKey(
  themeId: string,
  version: string,
  mode: string,
): ThemeCacheKey {
  return `${themeId}:${version}:${mode}`;
}

/**
 * LRU-bounded CompiledTheme store (THEME-09).
 * - Map iteration order = recency order (first key = LRU).
 * - get() refreshes recency (delete + re-set).
 * - set() evicts the LRU entry when size exceeds maxEntries.
 * - Explicit invalidate/clear only — NO auto-invalidation wiring
 *   in Phase 3 (03-CONTEXT): invalidate(themeId?, version?, mode?)
 *   with a partial argument matches on the PROVIDED fields only.
 */
export class ThemeCache {
  private readonly cache = new Map<ThemeCacheKey, CompiledTheme>();
  constructor(private readonly maxEntries: number = THEME_CACHE_DEFAULT_MAX) {}

  get(
    themeId: string,
    version: string,
    mode: string,
  ): CompiledTheme | undefined;
  set(
    themeId: string,
    version: string,
    mode: string,
    compiled: CompiledTheme,
  ): void;
  has(themeId: string, version: string, mode: string): boolean;
  /** Remove matching keys. Missing args act as wildcards; no args clears all. */
  invalidate(themeId?: string, version?: string, mode?: string): void;
  /** Empty the cache entirely. */
  clear(): void;
  /** Number of cached entries. */
  get size(): number;
}
```

Semantics locked by tests:

- Key composition: `toThemeCacheKey("ocean", "1.0.0", "dark")` →
  `"ocean:1.0.0:dark"`; `"ocean"` + `"1.0.0"` + `"system"` is a DIFFERENT key.
- `get` on a present key returns the SAME object reference stored by
  `set` (identity, not a copy — the runtime hands the reference to the
  injector; determinism lives in the compiler, not the cache).
- LRU: `set` beyond maxEntries evicts the first Map key (least-recently
  used); `get` refreshes recency so a touched key survives eviction.
- `invalidate("ocean", undefined, undefined)` removes ALL "ocean" keys;
  `invalidate("ocean", "1.0.0", "dark")` removes exactly that one;
  zero-arg `invalidate()` == `clear()`.
- No auto-invalidation: nothing subscribes to events in this module
(zero event-bus/theme.changed references — grep gate below).

| --------------- | ---------------------- | --------------------------------------- | ----------- | --------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| T3-05           | Tampering              | `set()`/`get()` key handling            | mitigate    | Composite key built from three explicit parts with a ":" separator; eviction and invalidation only ever operate on whole keys — no partial overwrite, no key-format injection (a version/mode containing ":" splits deterministically and can only match its own composite) |
| T3-06           | Denial of service      | unbounded growth                        | mitigate    | Hard LRU cap (default 100) enforced on every set(); while-loop eviction guarantees size ≤ maxEntries invariant at all times                                                                                                                                                 |
| T3-07           | Information disclosure | cached compiled values                  | accept      | CompiledTheme is a public string map of CSS variables (non-secret design tokens); the cache is process-local in-memory state — no serialization, no persistence surface in this phase                                                                                       |
| T3-08           | Spoofing               | stale entries served after invalidation | accept      | Phase 3 deliberately ships manual invalidation only (03-CONTEXT); stale-entry risk is bounded to explicit caller behavior and is the documented Phase-4 auto-subscription point                                                                                             |
| </threat_model> |
